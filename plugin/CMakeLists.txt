cmake_minimum_required(VERSION 3.21)
project(NBXardasAI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(GOTHIC_SDK_DIR "${CMAKE_SOURCE_DIR}/../third_party" CACHE PATH "Folder z union-api i gothic-api (gitlab.com/union-framework)")
set(GENERATE_UNION_API_LIB ON CACHE BOOL "" FORCE)
add_subdirectory("${GOTHIC_SDK_DIR}/union-api" union-api-build)
add_library(NBXardasAI SHARED plugin.cpp)
set_target_properties(NBXardasAI PROPERTIES PREFIX "")
target_include_directories(NBXardasAI PRIVATE "${GOTHIC_SDK_DIR}/gothic-api" "${GOTHIC_SDK_DIR}/gothic-api/ZenGin/Gothic_UserAPI" "${GOTHIC_SDK_DIR}/union-api/dependencies/detours/src")
target_link_directories(NBXardasAI PRIVATE "${GOTHIC_SDK_DIR}/gothic-api")
target_compile_definitions(NBXardasAI PRIVATE _CRT_SECURE_NO_WARNINGS __G2A WIN32_LEAN_AND_MEAN)
target_compile_options(NBXardasAI PRIVATE /W4 /EHa)
target_link_libraries(NBXardasAI PRIVATE union_api_lib bcrypt winmm)
add_executable(NBXardasProtocolTests tests/protocol_tests.cpp)
target_link_libraries(NBXardasProtocolTests PRIVATE bcrypt)

add_executable(NBXardasAbiTests tests/abi_tests.cpp)

add_executable(NBXardasWaveTests tests/wave_tests.cpp)
target_link_libraries(NBXardasWaveTests PRIVATE winmm)
add_executable(NBXardasStreamTraceTests tests/stream_trace_tests.cpp)
target_link_libraries(NBXardasStreamTraceTests PRIVATE winmm)

foreach(t NBXardasProtocolTests NBXardasAbiTests NBXardasWaveTests NBXardasStreamTraceTests)
  target_include_directories(${t} PRIVATE ${CMAKE_SOURCE_DIR})
endforeach()
